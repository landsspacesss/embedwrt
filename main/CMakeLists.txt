# Build provenance, in a generated header.
#
# Generated by a build step rather than by CMake at configure time: whether the
# tree is dirty changes when a source file is edited, and CMake does not
# re-configure for that, so a configure-time value would go stale in exactly the
# case the guard exists for.
#
# A header rather than compile definitions, for a related reason: definitions
# have to be threaded through the cache to reach a component (add_compile_definitions
# at the top level does not), and getting that wrong fails silently - the macro
# stays undefined and the guard simply never fires.
#
# fw_update.c includes it via #include "fw_build_info.h".
set(EMBEDWRT_BUILD_INFO_DIR "${CMAKE_BINARY_DIR}/generated")
set(EMBEDWRT_BUILD_INFO "${EMBEDWRT_BUILD_INFO_DIR}/fw_build_info.h")

add_custom_target(embedwrt_build_info ALL
    COMMAND ${CMAKE_COMMAND} -E make_directory "${EMBEDWRT_BUILD_INFO_DIR}"
    COMMAND ${CMAKE_COMMAND}
            -DSRC=${CMAKE_SOURCE_DIR}
            -DOUT=${EMBEDWRT_BUILD_INFO}
            -P ${CMAKE_SOURCE_DIR}/tools/gen_build_info.cmake
    BYPRODUCTS "${EMBEDWRT_BUILD_INFO}"
    COMMENT "Recording build provenance")

add_dependencies(${COMPONENT_LIB} embedwrt_build_info)
target_include_directories(${COMPONENT_LIB} PRIVATE "${EMBEDWRT_BUILD_INFO_DIR}")
